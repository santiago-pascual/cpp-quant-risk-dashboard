cmake_minimum_required(VERSION 3.16)
project(risk_dashboard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 add_compile_options(/W4)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(risk src/analytics.cpp src/csv.cpp)
target_include_directories(risk PUBLIC include)
add_executable(dashboard src/main.cpp)
target_link_libraries(dashboard PRIVATE risk)
enable_testing()
add_executable(risk_tests tests/test.cpp)
target_link_libraries(risk_tests PRIVATE risk)
add_test(NAME risk_tests COMMAND risk_tests)
